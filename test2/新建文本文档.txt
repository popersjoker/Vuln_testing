-- 源码文件信息表--
DROP TABLE IF EXISTS `table_file`;
CREATE TABLE `table_file`(
	`yid` int(0) NOT NULL AUTO_INCREMENT COMMENT '源码文件id',
	`file_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL DEFAULT '' COMMENT '源码文件名',
	`upload_time` datetime(0) NOT NULL COMMENT '上传时间',
	`upload_url` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL DEFAULT '' COMMENT '源码文件上传位置',
	PRIMARY KEY(`yid`) 
)ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='源码文件信息表';	



-- 编译记录表--
DROP TABLE IF EXISTS `table_compile`;
CREATE TABLE `table_compile`(
	`cid` int(0) NOT NULL AUTO_INCREMENT COMMENT '编译文件id',
	`compiled_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL DEFAULT '' COMMENT '编译文件名',
	`compiled_url` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL DEFAULT '' COMMENT '编译文件路径',
	`yid` int(0) NOT NULL COMMENT '源码文件id',
	PRIMARY KEY(`cid`),
	FOREIGN KEY(`yid`) REFERENCES `table_file` (`yid`) ON DELETE CASCADE
)ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='编译记录表';
	


-- 输入文件上传记录表--
DROP TABLE IF EXISTS `table_upload`;
CREATE TABLE `table_upload`(
	`uid` int(0) NOT NULL AUTO_INCREMENT COMMENT '输入文件id',
	`input_file_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '输入文件名',
	`upload_input_url` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '上传输入文件路径',
	`yid` int(0) NOT NULL COMMENT '源码文件id',
	PRIMARY KEY(`uid`) USING BTREE,	
	INDEX `FK_file`(`yid`)USING BTREE,
	CONSTRAINT `FK_file` FOREIGN KEY(`yid`)REFERENCES `table_file` (`yid`)ON DELETE CASCADE ON UPDATE CASCADE
)ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='输入文件上传记录表';



-- 测试记录表--
DROP TABLE IF EXISTS `table_record`;
CREATE TABLE `table_record`(
	`fid` int(0) NOT NULL AUTO_INCREMENT COMMENT '模糊测试记录id',
	`test_time` datetime(0) NULL DEFAULT NULL COMMENT '测试时间',
	`fuzz_out_url` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '输出文件url',
	`yid` int(0) NOT NULL COMMENT '源码文件id',
	`cid` int(0) NOT NULL COMMENT '编译文件id',
	`uid` int(0) NOT NULL COMMENT '输入文件id',
	`uuid` varchar(255)CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '唯一识别标识',
	PRIMARY KEY(`fid`),
	FOREIGN KEY(`yid`) REFERENCES `table_file` (`yid`)  ON DELETE CASCADE,
	FOREIGN KEY(`cid`) REFERENCES `table_compile` (`cid`) ON DELETE CASCADE,
	FOREIGN KEY(`uid`) REFERENCES `table_upload` (`uid`)
)ENGINE=INNODB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='测试记录表';
