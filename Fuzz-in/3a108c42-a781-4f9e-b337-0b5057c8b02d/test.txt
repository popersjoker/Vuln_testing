from transformers import  AutoModel,AutoTokenizer
import pandas as pd
import numpy as np
import datasets
from tqdm import tqdm 

model_name = "chatglm2-6b" 
tokenizer = AutoTokenizer.from_pretrained(
    model_name, trust_remote_code=True)
model = AutoModel.from_pretrained(model_name,trust_remote_code=True).half().cuda()
# 要想利用大模型实现分类任务，要先引导它做规范的预测，不然会变成对话情况，比如给它一句差评，它不会直接说，而是解释原因或者道歉
prompt = """文本分类任务：将一段用户给外卖服务的评论进行分类，分成好评或者差评。

下面是一些范例:

味道真不错 -> 好评
太辣了，吃不下都  -> 差评

请对下述评论进行分类。返回'好评'或者'差评'，无需其它说明和解释。

xxxxxx ->

"""

def get_prompt(text):
    return prompt.replace('xxxxxx',text)
# 测试效果，未引导前
response, his = model.chat(tokenizer, get_prompt('味道不错'), history=[])
# print(response)
  
# 增加4个范例，目的就是人为地加入一些上下文，引导之后，模型的输出就是好评或差评了
his.append(("太贵了 -> ","差评"))
his.append(("非常快，味道好 -> ","好评"))
his.append(("这么咸真的是醉了 -> ","差评"))
his.append(("价格感人 优惠多多 -> ","好评"))

# 将输入输出封装成一个函数
def predict(text):
    response, history = model.chat(tokenizer, f"{text} ->", history=his,
    temperature=0.01)
    return response
 
pd.set_option('display.max_colwidth', None) # 取消文本长度限制，当然这里输出只有好评or差评2个字符，长文本输出时需要加这一句
df = pd.read_csv("waimai_10k.csv") # 数据集数据格式：label,review
df['tag'] = df['label'].map({0:'差评',1:'好评'}) # 现在label = 0/1 对应tag = 差评/好评
df = df.rename({'review':'text'},axis=1)  # text = review
dfgood = df.query('tag=="好评"')
dfbad = df.query('tag=="差评"').head(len(dfgood))
df = pd.concat([dfgood,dfbad]) # 从数据集中拿到相同数量的好评行和差评行 example:好评4000条，对应拿到的差评数也是4000
ds_dic = datasets.Dataset.from_pandas(df).train_test_split(
    test_size = 2000, shuffle=True, seed = 43)  # 随机抽出2000条评论
dftest = ds_dic['test'].to_pandas() 
dftest.to_parquet('data/dftest.parquet')  # 数据集中的数据保存为parquet列式存储格式，方便取数据
preds = ['' for x in dftest['tag']] 
for i in tqdm(range(len(dftest))):
    text = dftest['text'].loc[i] # text就是评论
    preds[i] = predict(text)  # preds[i] = 对于该评论，模型给出的评价
dftest['pred'] = preds 
acc = len(dftest.query('tag==pred'))/len(dftest)  # 计算分类准确率，模型预测==对应数据集中该评论的真实评价 的数量  / 用来测试的评论总数
dftest[['text', 'tag', 'pred']].to_csv('my_output.csv', sep=',', index=False) # 运行结果写入csv文件
df = pd.DataFrame(['说明:text是数据集中的评论,tag是在数据集中该评论对应的评价(好评or差评),pred是模型对该评论产生的评价(好评or差评)'],columns=['text'])
df.to_csv('my_output.csv', mode='a', header=False, index=False)
df = pd.DataFrame([f'Model Accuary:{acc}'], columns=['text'])
df.to_csv('my_output.csv', mode='a', header=False, index=False)  # 准确率写入csv文件末尾
